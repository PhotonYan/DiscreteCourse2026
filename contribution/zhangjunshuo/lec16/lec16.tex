\section*{一页式总览}
\begin{itemize}
\item 条件概率：$P(A\mid B)=P(A\cap B)/P(B)$；条件分布 $p_B(\omega)=p(\omega)/P(B)$（$\omega\in B$）。
\item Bayes：$P(B_i\mid A)=P(B_i)P(A\mid B_i)/\sum_jP(B_j)P(A\mid B_j)$。
\item 独立事件：$P(A\cap B)=P(A)P(B)$；独立随机变量：联合分布等于边缘分布之积。
\item 期望：$E[X]=\sum_\omega P(\omega)X(\omega)=\sum_x x p_X(x)$；线性性；独立时 $E[XY]=E[X]E[Y]$。
\item 全期望：$E[X]=E[E[X\mid Y]]=\sum_yP(Y=y)E[X\mid Y=y]$。
\item 方差：$\operatorname{Var}(X)=E[X^2]-(E[X])^2$；$\operatorname{Var}(X+Y)=\operatorname{Var}X+\operatorname{Var}Y+2\operatorname{Cov}(X,Y)$。
\item 协方差：$\operatorname{Cov}(X,Y)=E[XY]-E[X]E[Y]$；独立 $\Rightarrow$ 协方差为 $0$，反之不成立。
\item PGF：$G_X(z)=E[z^X]=\sum_kp_kz^k$；$G_X(1)=1$，$G_X'(1)=E[X]$，$G_X''(1)=E[X(X-1)]$。
\item 独立和：$G_{X+Y}=G_XG_Y$。
\item 常见 PGF：Bernoulli $1-p+pz$；二项 $(1-p+pz)^n$；几何 $pz/(1-(1-p)z)$；Poisson $e^{\lambda(z-1)}$。
\item Poisson 极限：$\operatorname{Bin}(n,\lambda/n)\Rightarrow\operatorname{Pois}(\lambda)$。
\end{itemize}
